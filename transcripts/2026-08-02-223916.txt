  Step 1 — flat struct that mirrors the CSV columns:

  #[derive(Deserialize)]
  struct RawEvent {
      #[serde(rename = "type")]
      ty: String,
      client: ClientID,
      tx: TxID,
      amount: Option<Amount>,  // optional because Dispute/Resolve have no 
  amount
  }

  Step 2 — convert with TryFrom:

  impl TryFrom<RawEvent> for Event {
      type Error = String;

      fn try_from(r: RawEvent) -> Result<Self, Self::Error> {
          match r.ty.trim() {
              "deposit" => Ok(Event::Deposit {
                  client: r.client,
                  tx: r.tx,
                  amount: r.amount.ok_or("deposit missing amount")?,
              }),
              "withdrawal" => Ok(Event::Withdrawal {
                  client: r.client,
                  tx: r.tx,
                  amount: r.amount.ok_or("withdrawal missing amount")?,
              }),
              "dispute" => Ok(Event::Dispute { client: r.client, tx: r.tx }),
              "resolve" => Ok(Event::Resolve { client: r.client, tx: r.tx }),
              other => Err(format!("unknown event type: {other}")),
          }
      }
  }
